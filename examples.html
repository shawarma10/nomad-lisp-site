    Example Programs

    Short programs that demonstrate the language.

    More examples are available in the examples directory of the repository.

    Fibonacci

    (letfun fib (n)
  (switch n
    (0 0)
    (1 1)
    (_ (+ (fib (dec n)) (fib (- n 2))))))

(println (fib 20))

    Factorial

    (letfun fact (n)
  (switch n
    (0 1)
    (_ (* n (fact (dec n))))))

(println (fact 10))

    Closures and currying

    (let add
  (lambda (x)
    (lambda (y)
      (+ x y))))

(let add10 (add 10))
(let x (add10 20))
(println "x = " x)
(if (= x 30)
  (println "All good! Closures work as expected.")
  (println "Uh oh, Closures do not work as expected!"))

    A quiz

    (letfun input_loop (correct_answer prompt)
  (if (= correct_answer (readln prompt))
    (println "Correct!")
    (do
      (println "False!\nTry again!")
      (input_loop correct_answer prompt))))

(input_loop "berlin" "What is the capital of Germany? ")
(input_loop "ocaml" "What language is Nomad Lisp written in? ")
(println "All questions answered correctly!")

    Records

    (let person
  (record
    (name "John Doe")
    (age 22)
    (job "Electrician")))

(println (. person name) " is " (. person age) " years old")
